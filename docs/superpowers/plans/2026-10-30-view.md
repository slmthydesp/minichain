# 链查看页 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 做一个单独的查看进程，页面上填多个出块进程的 `主机:端口`，并排画出各自的规范链。

**Architecture:** `view` 不导入 `chain`。它用 `CanonicalNode` 校验地址，把 `/api/status` 和 `/api/blocks` 转到 `http://<主机:端口>` 上现成的 `/status` 和 `/blocks`，并用 `go:embed` 提供 `view/static/index.html`。`cmd/view` 只解析 `--listen` 并调用 `view.Handler()`。浏览器把名单存在 `localStorage`，每秒向查看进程拉最近 24 块。

**Tech Stack:** Go 1.26.1 标准库（`net/http`、`go:embed`、`httptest`）。页面是一个无构建步骤的 HTML 文件，内联 CSS 和 JS。

**Spec:** `docs/superpowers/specs/2026-10-30-view-design.md`

## Global Constraints

- 新的依赖、节点改动、CORS、侧链、账户列表、交易池、交易签名的展示、自动化浏览器测试。测试不调用 `time.Sleep`。
- `view` 不导入 `chain`。它不解析区块，只转发 HTTP。
- 除了 `GET /`、`GET /api/status`、`GET /api/blocks`，其他路径返回 `404`。页面路由必须写成 `GET /{$}`，不能写成 `GET /`，否则会吞掉未注册的路径。
- `height` 是 JSON 数字，并且是 0 到 2^53−1 的整数（含）。更大的数在浏览器里不能精确表示，这轮算解析失败。
- 模块路径：`github.com/slmthydesp/minichain`。Go 1.26.1。不新增依赖，不修改 `chain/` 和 `node/`。
- 用户要求开发过程不提交。执行本计划时不要运行 `git commit` 或 `git add`。每个任务结束只运行测试，改动留在工作区。

把下面的文件写成 `view/addr.go`。先去掉两端空白，再 `net.SplitHostPort`。端口用 `strconv.ParseUint(port, 10, 16)`，结果为 0 也是 `bad node`。主机里出现空白、`@`、`/`、`\` 是 `bad node`。

```go
package view

import (
	"errors"
	"net"
	"strconv"
	"strings"
	"unicode"
)

func CanonicalNode(raw string) (string, error) {
	raw = strings.TrimSpace(raw)
	host, port, err := net.SplitHostPort(raw)
	if err != nil || host == "" {
		return "", errors.New("bad node")
	}
	for _, r := range host {
		if unicode.IsSpace(r) || r == '@' || r == '/' || r == '\\' {
			return "", errors.New("bad node")
		}
	}
	n, err := strconv.ParseUint(port, 10, 16)
	if err != nil || n == 0 {
		return "", errors.New("bad node")
	}
	if strings.Contains(host, ":") {
		return "[" + host + "]:" + strconv.FormatUint(n, 10), nil
	}
	return host + ":" + strconv.FormatUint(n, 10), nil
}
```

```go
package view

import (
	"io"
	"net/http"
	"strconv"
	"time"
)

var upstreamClient = &http.Client{
	Timeout: 3 * time.Second,
	CheckRedirect: func(*http.Request, []*http.Request) error {
		return http.ErrUseLastResponse
	},
}

func Handler() http.Handler {
	mux := http.NewServeMux()
	mux.HandleFunc("GET /api/status", handleStatus)
	mux.HandleFunc("GET /api/blocks", handleBlocks)
	return mux
}

func handleStatus(w http.ResponseWriter, r *http.Request) {
	node := r.URL.Query().Get("node")
	if _, err := CanonicalNode(node); err != nil {
		http.Error(w, "bad node", http.StatusBadRequest)
		return
	}
	forward(w, node, "/status")
}

func handleBlocks(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	node := q.Get("node")
	if _, err := CanonicalNode(node); err != nil {
		http.Error(w, "bad node", http.StatusBadRequest)
		return
	}
	path := "/blocks"
	if _, ok := q["after"]; ok {
		n, err := strconv.ParseUint(q.Get("after"), 10, 64)
		if err != nil {
			http.Error(w, "bad after", http.StatusBadRequest)
			return
		}
		path = "/blocks?after=" + strconv.FormatUint(n, 10)
	}
	forward(w, node, path)
}

func forward(w http.ResponseWriter, node, path string) {
	canonical, err := CanonicalNode(node)
	if err != nil {
		http.Error(w, "bad node", http.StatusBadRequest)
		return
	}
	// 不转发 Location，页面看到的才是上游的 3xx，而不是被浏览器带走。
	req, err := http.NewRequest(http.MethodGet, "http://"+canonical+path, nil)
	if err != nil {
		http.Error(w, "bad node", http.StatusBadRequest)
		return
	}
	resp, err := upstreamClient.Do(req)
	if err != nil {
		http.Error(w, "unreachable", http.StatusBadGateway)
		return
	}
	defer resp.Body.Close()
	body, err := io.ReadAll(resp.Body)
	if err != nil {
		http.Error(w, "unreachable", http.StatusBadGateway)
		return
	}
	if ct := resp.Header.Get("Content-Type"); ct != "" {
		w.Header().Set("Content-Type", ct)
	}
	w.WriteHeader(resp.StatusCode)
	_, _ = w.Write(body)
}
```

    function loadNodes() {
      try {
        const parsed = JSON.parse(localStorage.getItem(storageKey) || "[]");
        if (!Array.isArray(parsed) || parsed.some((item) => typeof item !== "string")) return [];
        return parsed;
      } catch (err) {
        return [];
      }

      localStorage.setItem(storageKey, JSON.stringify(nodes));

      return { inflight: false, error: "", status: null, blocks: null, openHash: "" };

    function canonicalNode(raw) {
      const trimmed = String(raw).trim();
      let host = "";
      let port = "";
      if (trimmed.startsWith("[")) {
        const end = trimmed.indexOf("]");
        if (end <= 1 || trimmed[end + 1] !== ":") return null;
        host = trimmed.slice(1, end);
        port = trimmed.slice(end + 2);
      } else {
        const i = trimmed.lastIndexOf(":");
        if (i <= 0) return null;
        host = trimmed.slice(0, i);
        port = trimmed.slice(i + 1);
        if (host.includes(":")) return null;
      }
      if (!host || /[@/\\\s]/.test(host)) return null;
      if (!/^[0-9]+$/.test(port)) return null;
      const n = Number(port);
      if (!Number.isSafeInteger(n) || n < 1 || n > 65535) return null;
      if (host.includes(":")) return "[" + host + "]:" + n;
      return host + ":" + n;
    }

    function isUint(n) {
      return Number.isSafeInteger(n) && n >= 0;
    }

    function validStatus(v) {
      return !!v && typeof v === "object" && !Array.isArray(v) &&
        isUint(v.height) && typeof v.stateRoot === "string" && typeof v.head === "string";
    }

    function validBlocks(v) {
      if (!Array.isArray(v)) return false;
      return v.every((block) => block && typeof block === "object" &&
        isUint(block.height) &&
        typeof block.parentHash === "string" &&
        typeof block.stateRoot === "string" &&
        typeof block.hash === "string" &&
        typeof block.coinbase === "string" &&
        Array.isArray(block.txs) &&
        block.txs.every((tx) => tx && typeof tx.to === "string" && isUint(tx.value) && isUint(tx.nonce)));
    }

    function abbrev(raw) {
      const original = String(raw);
      let body = original;
      if (body.startsWith("0x") || body.startsWith("0X")) body = body.slice(2);
      const shown = body.length >= 12 ? body.slice(0, 8) + "…" + body.slice(-4) : body;
      return { shown, original };
    }

    async function readJSON(res) {

        return { value: await res.json(), ok: true };

        return { value: null, ok: false };

    async function refresh(node) {
      const col = state.get(node);
      if (!col || col.inflight) return;
      col.inflight = true;
      try {
        let statusRes;
        try {
          statusRes = await fetch("/api/status?node=" + encodeURIComponent(node), { cache: "no-store" });
        } catch (err) {
          col.error = "连不上";
          render();
          return;
        }
        if (statusRes.status === 502) {
          col.error = "连不上";
          render();
          return;
        }
        if (!statusRes.ok) {
          col.error = String(statusRes.status);
          render();
          return;
        }
        const statusRead = await readJSON(statusRes);
        if (!statusRead.ok || !validStatus(statusRead.value)) {
          col.error = "回应无法解析";
          render();
          return;
        }
        const height = statusRead.value.height;
        const after = height <= 24 ? 0 : height - 24;
        let blocksRes;
        try {
          blocksRes = await fetch("/api/blocks?node=" + encodeURIComponent(node) + "&after=" + after, { cache: "no-store" });
        } catch (err) {
          col.error = "连不上";
          render();
          return;
        }
        if (blocksRes.status === 502) {
          col.error = "连不上";
          render();
          return;
        }
        if (!blocksRes.ok) {
          col.error = String(blocksRes.status);
          render();
          return;
        }
        const blocksRead = await readJSON(blocksRes);
        if (!blocksRead.ok || !validBlocks(blocksRead.value)) {
          col.error = "回应无法解析";
          render();
          return;
        }
        col.error = "";
        col.status = statusRead.value;
        col.blocks = blocksRead.value;
        if (col.openHash && !col.blocks.some((block) => block.hash === col.openHash)) col.openHash = "";
        render();
      } finally {
        col.inflight = false;
      }
    }

      state.delete(node);

      }
    }

package main

import (
	"flag"
	"fmt"
	"net/http"
	"os"

	"github.com/slmthydesp/minichain/view"

func main() {
	listen := flag.String("listen", "127.0.0.1:8080", "listen address")
	flag.Parse()
	if err := http.ListenAndServe(*listen, view.Handler()); err != nil {

		os.Exit(1)
	}
}
